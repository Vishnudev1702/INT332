name: CI-CD Pipeline

on:
  push:
    branches:
      - main

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Build Docker Image
        run: docker build -t my-docker-app .

      - name: Run Container (Deployment Step)
        run: docker run -d -p 5000:5000 my-docker-app

      - name: Check Running Containers
        run: docker ps
