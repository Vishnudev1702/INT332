
# Pull Maven image
docker pull maven:3.9.10-eclipse-temurin-17

# Create Maven project (first time only)
docker run --rm -it -v ${PWD}:/app -w /app maven:3.9.10-eclipse-temurin-17 mvn archetype:generate -DgroupId=com.example -DartifactId=myapp -DarchetypeArtifactId=maven-archetype-quickstart -DinteractiveMode=false

# Go into project
cd myapp

# Build project using Docker
docker run --rm -it -v ${PWD}:/app -w /app maven:3.9.10-eclipse-temurin-17 mvn clean install

# Run the Java application
java -cp target/myapp-1.0-SNAPSHOT.jar com.example.App

# Open docker-compose file
open -e docker-compose.yaml

# Run using Docker Compose
docker compose up --build

# Stop and remove containers
docker compose down

# Remove conflicting containers (when needed)
docker stop mysql-db
docker rm mysql-db
docker stop wordpress-app
docker rm wordpress-app

# Force remove containers (quick fix)
docker rm -f mysql-db wordpress-app

# Check running containers
docker ps

# Check all containers
docker ps -a



version: "3.9"
